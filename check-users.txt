import { db } from "./db/index.js";
import { sql } from "drizzle-orm";

async function run() {
  try {
    const res = await db.execute(sql`SELECT * FROM hero_users LIMIT 1;`);
    console.log("Users OK:", res.rows.length);
  } catch(e) {
    console.error("Users Error:", e.message);
  }
  process.exit(0);
}
run();
