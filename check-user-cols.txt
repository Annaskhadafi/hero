import { db } from "./db/index.js";
import { sql } from "drizzle-orm";

async function run() {
  try {
    const res = await db.execute(sql`SELECT column_name, data_type FROM information_schema.columns WHERE table_name = 'user';`);
    console.log("User columns:", res.rows);
  } catch(e) {
    console.error(e.message);
  }
  process.exit(0);
}
run();
